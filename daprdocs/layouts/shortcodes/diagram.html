{{- /* {{< diagram overview >}} Inlines daprdocs/assets/diagrams/.svg, which `npm run export:diagrams` generates from diagrams/scenes/.mjs. Inlined rather than linked so the var(--d-*) colours inside it resolve against the page's theme tokens and follow the dark-mode toggle with no JavaScript. A missing or misspelled name fails the build instead of rendering nothing. */ -}} {{- $name := or (.Get 0) (.Get "name") -}} {{- with $name -}} {{- /* Same rule the exporter enforces on a scene name (diagrams/export.mjs). Without it the argument is interpolated straight into a resource path, so a typo can reach for any asset and inline it through safeHTML. */ -}} {{- if not (findRE `^[a-z0-9]+(-[a-z0-9]+)*$` . ) -}} {{- errorf "diagram shortcode: %q is not a scene name - lowercase letters, digits and single hyphens only (called from %s)" . $.Position -}} {{- end -}} {{- $path := printf "diagrams/%s.svg" . -}} {{- with resources.Get $path -}} {{- else -}} {{- errorf "diagram shortcode: no diagram at assets/%s - run `npm run export:diagrams` or check the scene name (called from %s)" $path $.Position -}} {{- end -}} {{- else -}} {{- errorf "diagram shortcode: a scene name is required, as in {{< diagram overview >}} (called from %s)" .Position -}} {{- end -}}
